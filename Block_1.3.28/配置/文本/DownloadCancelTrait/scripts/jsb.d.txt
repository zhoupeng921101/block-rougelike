    /**
     * 下载客户端类
     *
     * @description 提供文件和数据下载功能，支持断点续传、优先级控制、进度回调等特性
     *
     * @example
     * ```javascript
     * const client = new jsb.DownloadClient({
     *     maxConcurrentTasks: 5,
     *     userAgent: "MyApp/1.0",
     *     enableResume: true
     * });
     *
     * const taskId = client.downloadFile({
     *     url: "https://example.com/file.zip",
     *     path: "/path/to/save",
     *     success: (taskId, url, path, totalBytes, elapsedMs) => {
     *         console.log(`下载完成: ${path}`);
     *     },
     *     failure: (taskId, url, code, httpStatus, msg) => {
     *         console.error(`下载失败: ${msg}`);
     *     },
     *     progress: (taskId, url, receivedBytes, total) => {
     *         console.log(`进度: ${(receivedBytes/total*100).toFixed(1)}%`);
     *     }
     * });
     * ```
     */
    export class DownloadClient {

        /**
         * 创建下载客户端实例
         * @param config 可选的客户端配置
         */
        constructor(config?: DownloadClient.ClientConfig);

        /**
         * 下载错误码枚举
         *
         * @description 定义所有可能的下载错误类型
         */
        static ErrorCode: {
            NONE: 0;
            INVALID_URL: -1;
            INVALID_PATH: -2;
            INVALID_PARAMS: -3;
            NETWORK: -4;
            TIMEOUT: -5;
            CANCELLED: -6;
            FILE_IO: -7;
            DISK_FULL: -8;
            PERMISSION: -9;
            HTTP_4XX: -10;
            HTTP_5XX: -11;
            OUT_OF_MEMORY: -12;
            UNKNOWN: -99;
        };

        /**
         * 下载文件到本地磁盘
         *
         * @param options 下载选项，包括 URL、保存路径、回调函数等
         * @returns 任务 ID，用于标识和控制下载任务；如果创建失败返回 -1
         *
         * @description
         * - 支持断点续传（通过 enableResume 选项）
         * - 支持自定义 HTTP 请求头
         * - 支持优先级控制
         * - 所有回调函数都在主线程（Cocos 线程）执行
         */
        downloadFile(options: DownloadClient.DownloadFileOption): number;

        /**
         * 下载数据到内存
         *
         * @param options 下载选项，包括 URL、回调函数等
         * @returns 任务 ID，用于标识和控制下载任务；如果创建失败返回 -1
         *
         * @description
         * - 数据以 ArrayBuffer 形式返回
         * - 适用于小文件或需要直接处理数据的场景
         * - 不支持断点续传
         * - 所有回调函数都在主线程（Cocos 线程）执行
         */
        downloadData(options: DownloadClient.DownloadDataOption): number;

        /**
         * 取消下载任务
         *
         * @param taskId 要取消的任务 ID
         * @param removeTmpFile 是否删除临时文件（默认 false）
         * @returns 是否成功取消任务
         *
         * @description
         * - 如果任务不存在或已完成，返回 false
         * - 取消后会触发 fail 回调，错误码为 ERROR_CANCELLED
         */
        abort(taskId: number, removeTmpFile?: boolean): boolean;
    }

    export namespace DownloadClient {

        /**
         * 错误码值类型（用于类型安全的错误码比较）
         *
         * @example
         * ```typescript
         * function handleError(code: jsb.DownloadClient.ErrorCodeValue) {
         *     if (code === jsb.DownloadClient.ErrorCode.ERROR_NETWORK) {
         *         console.log("网络错误");
         *     }
         * }
         * ```
         */
        export type ErrorCodeValue = typeof DownloadClient.ErrorCode[keyof typeof DownloadClient.ErrorCode];

        /**
         * 客户端配置选项
         */
        export interface ClientConfig {
            /** 最大并发任务数（默认: 6） */
            maxConcurrentTasks?: number;
            /** User-Agent 字符串（默认: "DownloadClient/2.0"） */
            userAgent?: string;
            /** 是否启用断点续传（默认: true） */
            enableResume?: boolean;
            /** 内存缓存最大大小，单位字节（默认: 100MB） */
            maxMemoryCacheSize?: number;
        }

        /**
         * 文件下载选项
         */
        export interface DownloadFileOption {
            /** 下载 URL（必填，不能为空） */
            url: string;
            /** 本地存储路径（必填，不能为空） */
            path: string;
            /** 自定义 HTTP 请求头（可选） */
            headers?: Record<string, string>;
            /** 超时时间（秒），0 表示使用默认值（可选） */
            timeoutSeconds?: number;
            /** 优先级：-1=低，0=默认，1-5=高（值越大优先级越高，可选，当前仅 iOS 生效，Android 为 FIFO 模式） */
            /*priority?: number;*/
            /** 是否启用断点续传（可选，默认继承客户端配置） */
            enableResume?: boolean;
            /**
             * 下载成功回调（必填）
             * @param taskId 任务 ID
             * @param requestURL 请求的 URL
             * @param storagePath 存储路径
             * @param totalBytes 总字节数
             * @param elapsedMs 总耗时（毫秒）
             */
            success: (taskId: number, requestURL: string, storagePath: string, totalBytes: number, elapsedMs: number) => void;
            /**
             * 下载失败回调（必填）
             * @param taskId 任务 ID
             * @param requestURL 请求的 URL
             * @param errorCode 错误码（DownloadClient.ErrorCode）
             * @param httpStatus HTTP 状态码（如适用）
             * @param errorStr 错误信息
             */
            failure: (taskId: number, requestURL: string, errorCode: number, httpStatus: number, errorStr: string) => void;
            /**
             * 下载进度回调（必填）
             * @param taskId 任务 ID
             * @param requestURL 请求的 URL
             * @param receivedBytes 已接收字节数（累计）
             * @param totalBytes 总字节数（期望）
             */
            progress: (taskId: number, requestURL: string, receivedBytes: number, totalBytes: number) => void;
        }

        /**
         * 数据下载选项
         */
        export interface DownloadDataOption {
            /** 下载 URL（必填，不能为空） */
            url: string;
            /** 自定义 HTTP 请求头（可选） */
            headers?: Record<string, string>;
            /** 超时时间（秒），0 表示使用默认值（可选） */
            timeoutSeconds?: number;
            /** 优先级：-1=低，0=默认，1-5=高（值越大优先级越高，可选） */
            priority?: number;
            /**
             * 下载成功回调（必填）
             * @param taskId 任务 ID
             * @param requestURL 请求的 URL
             * @param data 下载的数据（ArrayBuffer）
             * @param totalBytes 总字节数
             * @param elapsedMs 总耗时（毫秒）
             */
            success: (taskId: number, requestURL: string, data: ArrayBuffer, totalBytes: number, elapsedMs: number) => void;
            /**
             * 下载失败回调（必填）
             * @param taskId 任务 ID
             * @param requestURL 请求的 URL
             * @param errorCode 错误码（DownloadClient.ErrorCode）
             * @param httpStatus HTTP 状态码（如适用）
             * @param errorStr 错误信息
             */
            failure: (taskId: number, requestURL: string, errorCode: number, httpStatus: number, errorStr: string) => void;
            /**
             * 下载进度回调（必填）
             * @param taskId 任务 ID
             * @param requestURL 请求的 URL
             * @param receivedBytes 已接收字节数（累计）
             * @param totalBytes 总字节数（期望）
             */
            progress: (taskId: number, requestURL: string, receivedBytes: number, totalBytes: number) => void;
        }
    }

        /**
         *  Checks whether the path is an absolute path.
         *
         *  @note On Android, if the parameter passed in is relative to "@assets/", this method will treat it as an absolute path.
         *        Also on Blackberry, path starts with "app/native/Resources/" is treated as an absolute path.
         *
         *  @param path The path that needs to be checked.
         *  @return True if it's an absolute path, false if not.
         */
        export function isAbsolutePath (path:string):boolean;
        /** Returns the fullpath for a given filename.

        First it will try to get a new filename from the "filenameLookup" dictionary.
        If a new filename can't be found on the dictionary, it will use the original filename.
        Then it will try to obtain the full path of the filename using the FileUtils search rules: resolutions, and search paths.
        The file search is based on the array element order of search paths and resolution directories.

        For instance:

            We set two elements("/mnt/sdcard/", "internal_dir/") to search paths vector by setSearchPaths,
            and set three elements("resources-ipadhd/", "resources-ipad/", "resources-iphonehd")
            to resolutions vector by setSearchResolutionsOrder. The "internal_dir" is relative to "Resources/".

            If we have a file named 'sprite.png', the mapping in fileLookup dictionary contains `key: sprite.png -> value: sprite.pvr.gz`.
            Firstly, it will replace 'sprite.png' with 'sprite.pvr.gz', then searching the file sprite.pvr.gz as follows:

                /mnt/sdcard/resources-ipadhd/sprite.pvr.gz      (if not found, search next)
                /mnt/sdcard/resources-ipad/sprite.pvr.gz        (if not found, search next)
                /mnt/sdcard/resources-iphonehd/sprite.pvr.gz    (if not found, search next)
                /mnt/sdcard/sprite.pvr.gz                       (if not found, search next)
                internal_dir/resources-ipadhd/sprite.pvr.gz     (if not found, search next)
                internal_dir/resources-ipad/sprite.pvr.gz       (if not found, search next)
                internal_dir/resources-iphonehd/sprite.pvr.gz   (if not found, search next)
                internal_dir/sprite.pvr.gz                      (if not found, return "sprite.png")

            If the filename contains relative path like "gamescene/uilayer/sprite.png",
            and the mapping in fileLookup dictionary contains `key: gamescene/uilayer/sprite.png -> value: gamescene/uilayer/sprite.pvr.gz`.
            The file search order will be:

                /mnt/sdcard/gamescene/uilayer/resources-ipadhd/sprite.pvr.gz      (if not found, search next)
                /mnt/sdcard/gamescene/uilayer/resources-ipad/sprite.pvr.gz        (if not found, search next)
                /mnt/sdcard/gamescene/uilayer/resources-iphonehd/sprite.pvr.gz    (if not found, search next)
                /mnt/sdcard/gamescene/uilayer/sprite.pvr.gz                       (if not found, search next)
                internal_dir/gamescene/uilayer/resources-ipadhd/sprite.pvr.gz     (if not found, search next)
                internal_dir/gamescene/uilayer/resources-ipad/sprite.pvr.gz       (if not found, search next)
                internal_dir/gamescene/uilayer/resources-iphonehd/sprite.pvr.gz   (if not found, search next)
                internal_dir/gamescene/uilayer/sprite.pvr.gz                      (if not found, return "gamescene/uilayer/sprite.png")

        If the new file can't be found on the file system, it will return the parameter filename directly.

        This method was added to simplify multiplatform support. Whether you are using cocos2d-js or any cross-compilation toolchain like StellaSDK or Apportable,
        you might need to load different resources for a given file in the different platforms.

        @since v2.1
        */
        export function fullPathForFilename (filename:string):string;
